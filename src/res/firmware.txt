// ============================================================
//  Hydroponics demo - ESP32
//  Sensors : XKC-Y25-NPN level, analog TDS, DS18B20
//  Output  : pump via P2003BDG MOSFET board, 15 min on / 45 min off
//  Display : SSD1306 128x32 OLED, I2C
//  Status  : 3 LEDs (heartbeat / pump / alarm)
// ============================================================

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <OneWire.h>
#include <DallasTemperature.h>

// ---------- mode ----------
const bool CALIBRATION_MODE = false;  // true = print raw mV, pump stays off

// ---------- pins ----------
const int PIN_LEVEL = 4;
const int PIN_TDS   = 34;             // ADC1 only - ADC2 breaks with WiFi
const int PIN_TEMP  = 5;              // needs 4.7k pull-up to 3V3
const int PIN_PUMP  = 25;

const int LED_OK    = 18;             // green  - heartbeat
const int LED_PUMP  = 19;             // yellow - pump running
const int LED_ALARM = 23;             // red    - alarm (21 is taken by SDA)

const int PIN_SDA = 21;
const int PIN_SCL = 22;

const uint8_t OLED_ADDR = 0x3C;       // confirm with an I2C scan

// ---------- EC calibration ----------
// Three-point fit against handheld meter, temperature-compensated to 25 C:
//   Vcomp 0.6703 ->  662 uS/cm
//   Vcomp 1.0349 -> 1164 uS/cm
//   Vcomp 1.6660 -> 2282 uS/cm
// Valid roughly 650-2300 uS/cm. Outside that span, do not trust it.
// Handheld was not verified against a reference solution.
const float EC_A = 396.3;
const float EC_B = 701.1;
const float EC_C = 14.0;

const float PPM_FACTOR = 0.5;         // match the handheld's scale

// ---------- timing ----------
const unsigned long PUMP_ON_MS    = 15UL * 60 * 1000;
const unsigned long PUMP_CYCLE_MS = 60UL * 60 * 1000;
const unsigned long SAMPLE_MS     = 30UL * 1000;

// ---------- targets (basil) ----------
const float TEMP_MIN = 18.0, TEMP_MAX = 24.0;
const float EC_MIN = 1000.0, EC_MAX = 1600.0;

// ---------- globals ----------
Adafruit_SSD1306 display(128, 32, &Wire, -1);
OneWire oneWire(PIN_TEMP);
DallasTemperature ds(&oneWire);

unsigned long lastSample = 0;
bool alarmActive = false;
bool haveDisplay = false;

// ============================================================
//  Sensors
// ============================================================

// Median of 30 samples. The TDS probe is driven with an AC excitation
// signal, so single reads catch arbitrary phases and produce outliers
// that would drag a mean around.
float medianVolts() {
  const int N = 30;
  int buf[N];
  for (int i = 0; i < N; i++) { buf[i] = analogReadMilliVolts(PIN_TDS); delay(5); }
  for (int i = 1; i < N; i++) {
    int k = buf[i], j = i - 1;
    while (j >= 0 && buf[j] > k) { buf[j + 1] = buf[j]; j--; }
    buf[j + 1] = k;
  }
  return buf[N / 2] / 1000.0;
}

// Returns NAN on failure rather than the library's -127, which is a
// plausible-looking number that would quietly poison the compensation.
float readTemp() {
  ds.requestTemperatures();
  float t = ds.getTempCByIndex(0);
  return (t == DEVICE_DISCONNECTED_C) ? NAN : t;
}

bool waterPresent() { return digitalRead(PIN_LEVEL) == LOW; }

float voltsToEC(float vc) { return EC_A * vc * vc + EC_B * vc + EC_C; }

// ============================================================
//  Display
// ============================================================

// dir: -1 down arrow (low), 0 check (in range), +1 up arrow (high)
void drawIndicator(int16_t x, int16_t y, int dir) {
  if (dir > 0) {
    display.fillTriangle(x + 4, y, x, y + 7, x + 8, y + 7, SSD1306_WHITE);
  } else if (dir < 0) {
    display.fillTriangle(x + 4, y + 7, x, y, x + 8, y, SSD1306_WHITE);
  } else {
    display.drawLine(x,     y + 4, x + 3, y + 7, SSD1306_WHITE);
    display.drawLine(x + 1, y + 4, x + 4, y + 7, SSD1306_WHITE);
    display.drawLine(x + 3, y + 7, x + 8, y,     SSD1306_WHITE);
    display.drawLine(x + 4, y + 7, x + 9, y,     SSD1306_WHITE);
  }
}

int rangeDir(float value, float lo, float hi) {
  if (value < lo) return -1;
  if (value > hi) return  1;
  return 0;
}

void showReadings(float t, float ec) {
  if (!haveDisplay) return;

  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);

  display.setTextSize(1);
  display.setCursor(0, 0);
  display.print("TEMP");
  display.setCursor(64, 0);
  display.print("EC uS/cm");

  display.setTextSize(2);

  display.setCursor(0, 14);
  if (isnan(t)) display.print("--.-");
  else          display.printf("%4.1f", t);

  display.setCursor(62, 14);
  display.printf("%4.0f", ec);

  if (!isnan(t)) drawIndicator(50, 16, rangeDir(t, TEMP_MIN, TEMP_MAX));
  drawIndicator(112, 16, rangeDir(ec, EC_MIN, EC_MAX));

  display.display();
}

void showCalibration(float v, float t) {
  if (!haveDisplay) return;

  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);

  display.setTextSize(1);
  display.setCursor(0, 0);
  display.print("CALIBRATION");

  display.setTextSize(2);
  display.setCursor(0, 12);
  display.printf("%6.4fV", v);

  display.setTextSize(1);
  display.setCursor(96, 24);
  display.printf("%4.1fC", t);

  display.display();
}

void showMessage(const char *a, const char *b) {
  if (!haveDisplay) return;
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(0, 4);
  display.print(a);
  display.setCursor(0, 18);
  display.print(b);
  display.display();
}

// ============================================================
//  Setup
// ============================================================

void setup() {
  Serial.begin(115200);

  // outputs off before anything else - pins float between reset and here
  pinMode(PIN_PUMP, OUTPUT);
  digitalWrite(PIN_PUMP, LOW);

  pinMode(LED_OK, OUTPUT);
  pinMode(LED_PUMP, OUTPUT);
  pinMode(LED_ALARM, OUTPUT);
  digitalWrite(LED_OK, LOW);
  digitalWrite(LED_PUMP, LOW);
  digitalWrite(LED_ALARM, LOW);

  pinMode(PIN_LEVEL, INPUT_PULLUP);
  analogSetPinAttenuation(PIN_TDS, ADC_11db);   // full 0-3.3V range
  ds.begin();

  Wire.begin(PIN_SDA, PIN_SCL);
  haveDisplay = display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);
  if (!haveDisplay) Serial.println("OLED not found - check address and wiring");
  else showMessage("Hydroponics", "starting...");

  Serial.println(CALIBRATION_MODE ? "calibration mode" : "running");
}

// ============================================================
//  Main
// ============================================================

void loop() {
  unsigned long now = millis();

  // ---- pump ----
  // The level check sits inside the same expression that drives the pin,
  // evaluated every iteration. There is no path where the pump runs
  // without a live water check. Preserve that if you refactor.
  if (!CALIBRATION_MODE) {
    bool wantPump = (now % PUMP_CYCLE_MS) < PUMP_ON_MS;
    digitalWrite(PIN_PUMP, (wantPump && waterPresent()) ? HIGH : LOW);
  }

  // ---- status LEDs, every iteration ----
  digitalWrite(LED_PUMP, digitalRead(PIN_PUMP));
  digitalWrite(LED_ALARM, alarmActive);
  digitalWrite(LED_OK, (now % 2000) < 100);   // short flash = loop alive

  // ---- sampling gate (rollover-safe) ----
  if (now - lastSample < SAMPLE_MS) return;
  lastSample = now;

  float t  = readTemp();
  float tc = isnan(t) ? 25.0 : t;
  float v  = medianVolts();
  float vc = v / (1.0 + 0.02 * (tc - 25.0));   // ~2 %/C, normalised to 25 C

  if (CALIBRATION_MODE) {
    Serial.printf("V=%.4f  Vcomp=%.4f  T=%.1f\n", v, vc, tc);
    showCalibration(v, tc);
    return;
  }

  float ec    = voltsToEC(vc);
  float ppm   = ec * PPM_FACTOR;
  bool  water = waterPresent();

  showReadings(t, ec);

  Serial.printf("T=%.1fC  EC=%.0f uS/cm  TDS=%.0f ppm  level=%s  pump=%s\n",
                tc, ec, ppm,
                water ? "OK" : "LOW",
                digitalRead(PIN_PUMP) ? "ON" : "OFF");

  // ---- alarms ----
  alarmActive = false;

  if (!water) {
    Serial.println("  ALARM: reservoir low, pump locked out");
    alarmActive = true;
  }
  if (isnan(t)) {
    Serial.println("  ALARM: temp sensor not responding");
    alarmActive = true;
  } else if (tc > TEMP_MAX) {
    Serial.println("  ALARM: too warm, root rot risk");
    alarmActive = true;
  } else if (tc < TEMP_MIN) {
    Serial.println("  ALARM: too cold for basil");
    alarmActive = true;
  }

  // ---- advisories (not alarms) ----
  if (ec > EC_MAX)      Serial.println("  ACTION: EC high, add plain water");
  else if (ec < EC_MIN) Serial.println("  ACTION: EC low, add nutrient");
}
